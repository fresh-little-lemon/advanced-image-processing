cmake_minimum_required(VERSION 3.16)
project(multifocus_cpu VERSION 1.0 LANGUAGES CXX)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(multifocus src/fusion.cpp)
target_compile_features(multifocus PUBLIC cxx_std_17)
target_include_directories(multifocus PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(multifocus PUBLIC ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(multifocus PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(multifocus_cli src/main.cpp)
target_link_libraries(multifocus_cli PRIVATE multifocus)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_fusion tests/test_fusion.cpp)
  target_link_libraries(test_fusion PRIVATE multifocus)
  add_test(NAME fusion_invariants COMMAND test_fusion)
  add_executable(fusion_probe tests/probe.cpp)
  target_link_libraries(fusion_probe PRIVATE multifocus)
endif()
add_executable(fusion_benchmark tests/benchmark.cpp)
target_link_libraries(fusion_benchmark PRIVATE multifocus)
