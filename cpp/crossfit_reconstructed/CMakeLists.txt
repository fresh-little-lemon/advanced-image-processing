cmake_minimum_required(VERSION 3.16)
project(crossfit_estimator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(crossfit src/crossfit.cpp)
target_include_directories(crossfit PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(crossfit PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_executable(crossfit_cli src/crossfit_cli.cpp)
target_link_libraries(crossfit_cli PRIVATE crossfit)
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE crossfit)
add_executable(benchmark bench/benchmark.cpp)
target_link_libraries(benchmark PRIVATE crossfit)
enable_testing()
add_test(NAME native COMMAND native_tests)
